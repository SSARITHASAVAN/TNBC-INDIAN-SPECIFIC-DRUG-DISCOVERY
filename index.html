# Stage 1: Fetch TNBC-active compounds from ChEMBL
from chembl_webresource_client.new_client import new_client
import pandas as pd

activity = new_client.activity
# TNBC-relevant targets: BRCA1, PARP1, EGFR, AR, PI3K, STAT3
TNBC_TARGETS = ['CHEMBL3592', 'CHEMBL3009', 'CHEMBL203',
                'CHEMBL1871', 'CHEMBL2095233', 'CHEMBL3779']

records = []
for tid in TNBC_TARGETS:
    res = activity.filter(target_chembl_id=tid,
                          standard_type='IC50',
                          standard_units='nM').only([
        'molecule_chembl_id', 'canonical_smiles',
        'standard_value', 'pchembl_value'])
    records.extend(list(res))

df = pd.DataFrame(records)
df['pIC50'] = -np.log10(df['standard_value'].astype(float) * 1e-9)
df.to_csv('tnbc_chembl_raw.csv', index=False)

TNBC-Atlas: An AI/ML-Driven Drug Discovery Platform for Triple-Negative Breast Cancer with Country-Specific Epidemiological Integration
TNBC-Atlas Consortium · 2024 · Preprint available on bioRxiv
We present TNBC-Atlas, an open-access, reproducible drug discovery platform integrating graph neural networks, molecular docking, and ADMET prediction to identify therapeutic candidates for triple-negative breast cancer. The platform screens 847 compounds across six validated targets (BRCA1, PARP1, EGFR, PI3Kα, STAT3, AR) and incorporates country-specific TNBC burden data from 47 nations to prioritize candidates based on regional genetic variant prevalence and drug accessibility. Validation against experimentally confirmed inhibitors demonstrates AUC of 0.942 and enrichment factor of 8.3 at 1% FDR.
